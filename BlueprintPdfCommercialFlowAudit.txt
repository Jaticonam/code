00017: const PDF_DESCRIPTION_MAX_LENGTH = 190;
00018: 
00019: const cleanText = (value?: string | null, fallback = "") =>
00020:   String(value || fallback).replace(/\s+/g, " ").trim();
00021: 
00022: const normalizeStatus = (status?: string) => cleanText(status).toLowerCase();
00023: 
00024: const isHiddenProduct = (product: Product) =>
00025:   normalizeStatus(product.status) === "oculto";
00026: 
00027: const getMainImage = (product: Product) => {
00028:   const productId = cleanText(product.id);
00029:   const optimizedPdfImage = cleanText(PDF_IMAGE_MANIFEST[productId]);
00030:   const catalogImage = cleanText(product.img);
00031: 
00032:   return optimizedPdfImage || catalogImage || PDF_PRODUCT_PLACEHOLDER_IMAGE;
00033: };
00034: 
00035: const getPrimaryPrice = (product: Product) => {
00036:   const offerPrice = Number(product.price_offer || 0);
00037:   const basePrice = Number(product.price_1 || 0);
00038: 
00039:   return offerPrice > 0 ? offerPrice : basePrice;
00040: };
00041: 
00042: const getStockLabel = (product: Product) => {
00043:   const status = normalizeStatus(product.status);
00044: 
00045:   if (status === "agotado") {
00046:     return "Agotado";
00047:   }
00048: 
00049:   if (typeof product.stock !== "number") {
00050:     return "Consultar stock";
00051:   }
00052: 
00053:   if (product.stock <= 0) {
00054:     return "Consultar stock";
00055:   }
00056: 
00057:   return `Stock: ${product.stock} und.`;
00058: };
00059: 
00060: const getShortDescription = (product: Product) => {
00061:   const description = cleanText(product.description);
00062: 
00063:   if (!description) {
00064:     return "";
00065:   }
00066: 
00067:   if (description.length <= PDF_DESCRIPTION_MAX_LENGTH) {
00068:     return description;
00069:   }
00070: 
00071:   return `${description.slice(0, PDF_DESCRIPTION_MAX_LENGTH - 3).trim()}...`;
00072: };
00073: 
00074: export const mapProductToPdfProduct = (product: Product): PdfProduct => ({
00075:   id: cleanText(product.id),
00076:   title: cleanText(product.title, "Producto Wooly"),
00077:   description: getShortDescription(product),
00078:   category: cleanText(product.category, "Sin categoría"),
00079:   image: getMainImage(product),
00080: 
00081:   price1: Number(product.price_1 || 0),
00082:   price3: product.price_3,
00083:   price12: product.price_12,
00084:   price50: product.price_50,
00085:   price100: product.price_100,
00086:   offerPrice: product.price_offer,
00087: 
00088:   primaryPrice: getPrimaryPrice(product),
00089:   stock: product.stock,
00090:   stockLabel: getStockLabel(product),
00091: 
00092:   status: product.status,
00093:   priority: Number(product.priority || 0),
00094: });
00095: 
00096: export const mapProductsToPdfProducts = (products: Product[]) =>
00097:   products
00098:     .filter((product) => !isHiddenProduct(product))
00099:     .map(mapProductToPdfProduct)
00100:     .sort((a, b) => b.priority - a.priority);
